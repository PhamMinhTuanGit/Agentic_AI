{

/* STP Detected a Root Guard Violation */
#define SMI_STP_ROOT_GUARD_VIOLATE_SET 1 << 1
 
/* STP Detected a BPDU filter Violation */

#define SMI_STP_BPDU_GUARD_VIOLATE_UNSET 1 << 3

};
