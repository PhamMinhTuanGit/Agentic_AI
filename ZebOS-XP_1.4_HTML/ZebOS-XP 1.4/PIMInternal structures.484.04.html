PIM Master Structure
The PIM master structure carries per-VR level instance information.
Definition
struct pim_master
{
/* Pointer to VR. */
struct ipi_vr *vr;
 
/* Pointer to globals. */
struct lib_globals *zg;
 
/* PIM VR Create Time. */
u_int32_t start_time;
 
};
