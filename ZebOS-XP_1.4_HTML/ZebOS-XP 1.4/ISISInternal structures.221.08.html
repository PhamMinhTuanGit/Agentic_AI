isis_packet
This data structure contains IS-IS packet-related information, and a pointer to the RAW packet data stream.
Definition
struct isis_packet
{
/* Pointer of next packet. */
struct isis_packet *next;
 
/* In/Out buffer. */
struct stream *buf;
 
/* TLV vector. */
vector tlvvec;
 
/* Source MAC address. */
u_char mac_src[ETHER_ADDR_LEN];
 
/* Level Index. */
u_char index;
};
